local function leerArchivo(ruta)
  local f = io.open(ruta, "r")
  if not f then return nil end
  local c = f:read("*a")
  f:close()
  return c
end

local function decodificar(ruta)
  local c = leerArchivo(ruta)
  if c and c ~= "" then
    local ok, d = pcall(cjson.decode, c)
    if ok and type(d) == "table" then return d end
  end
  return nil
end

local function leerJSON(ruta, def)
  return decodificar(ruta) or decodificar(ruta .. ".bak") or def
end

local function escribirJSON(ruta, datos)
  local ok, txt = pcall(cjson.encode, datos)
  if not ok then return false end
  local tmp = ruta .. ".tmp"
  local f = io.open(tmp, "w")
  if not f then return false end
  f:write(txt)
  f:close()
  local actual = File(ruta)
  if actual.exists() then
    local bak = File(ruta .. ".bak")
    if bak.exists() then bak.delete() end
    actual.renameTo(bak)
  end
  return File(tmp).renameTo(File(ruta))
end

local function sha256(txt)
  local ok, res = pcall(function()
    local md = MessageDigest.getInstance("SHA-256")
    local bytes = md.digest(JString(txt).getBytes("UTF-8"))
    local t = {}
    for i = 0, bytes.length - 1 do
      t[#t + 1] = string.format("%02x", (bytes[i] + 256) % 256)
    end
    return table.concat(t)
  end)
  if ok then return res end
  return nil
end

local function nuevaSal() return string.format("%08x%08x", math.random(0, 0x7fffffff), os.time()) end
local function hashPass(pw, sal) return sha256((sal or "") .. pw) or pw end
local function trim(s) return (tostring(s):gsub("^%s+", ""):gsub("%s+$", "")) end
local function ent(x) return math.floor(x + 0.5) end
local function azar(a, b) return math.random(ent(a), ent(b)) end

local EXTENSIONES = { "mp3", "ogg", "wav" }

local function tamano(ruta)
  local f = File(ruta)
  if f.exists() then return f.length() end
  return 0
end

local function audioValido(ruta)
  local f = io.open(ruta, "rb")
  if not f then return false end
  local h = f:read(4) or ""
  f:close()
  if h:sub(1, 3) == "ID3" or h == "OggS" or h == "RIFF" then return true end
  local b1, b2 = h:byte(1, 2)
  return b1 == 0xFF and b2 ~= nil and b2 >= 0xE0
end

local function rutaSonido(id)
  for _, ext in ipairs(EXTENSIONES) do
    local r = snd_path .. "/" .. id .. "." .. ext
    if tamano(r) > 0 then
      if audioValido(r) then return r end
      pcall(function() File(r).delete() end)
    end
  end
  return nil
end

local mp_bgm, mp_sfx, mp_sfx2
local pista_actual = nil
local function volumen() return config.volumen or 0.35 end

local function liberar(mp)
  if not mp then return end
  pcall(function() mp.stop() end)
  pcall(function() mp.release() end)
end

-- secundario = true usa un segundo canal, para que dos efectos suenen a la vez
local function reproducirEfecto(id, secundario)
  local ruta = rutaSonido(id)
  if not ruta then return end
  pcall(function()
    if secundario then liberar(mp_sfx2) else liberar(mp_sfx) end
    local mp = MediaPlayer()
    mp.setAudioStreamType(AudioManager.STREAM_MUSIC)
    mp.setDataSource(ruta)
    mp.setVolume(1.0, 1.0)
    mp.prepare()
    mp.start()
    if secundario then mp_sfx2 = mp else mp_sfx = mp end
  end)
end

local function detenerMusica()
  if mp_bgm then liberar(mp_bgm) mp_bgm = nil end
  pista_actual = nil
end

local function cambiarMusica(id_pista)
  if not config.musica then detenerMusica() return end
  if pista_actual == id_pista and mp_bgm ~= nil then return end
  detenerMusica()
  local ruta = rutaSonido(id_pista)
  if not ruta then return end
  pcall(function()
    mp_bgm = MediaPlayer()
    mp_bgm.setAudioStreamType(AudioManager.STREAM_MUSIC)
    mp_bgm.setDataSource(ruta)
    mp_bgm.setLooping(true)
    mp_bgm.setVolume(volumen(), volumen())
    mp_bgm.prepare()
    mp_bgm.start()
    pista_actual = id_pista
  end)
end

local function liberarAudio()
  detenerMusica()
  liberar(mp_sfx) mp_sfx = nil
  liberar(mp_sfx2) mp_sfx2 = nil
end

local function escribirWav(id, rate, dur, fn)
  local floor, char = math.floor, string.char
  local n = floor(rate * dur)
  local partes = {}
  for i = 0, n - 1 do
    local v = fn(i / rate)
    if v > 1 then v = 1 elseif v < -1 then v = -1 end
    local s = floor(v * 30000)
    if s < 0 then s = s + 65536 end
    partes[#partes + 1] = char(s % 256, floor(s / 256))
  end
  local datos = table.concat(partes)
  local function le32(x) return char(x % 256, floor(x / 256) % 256, floor(x / 65536) % 256, floor(x / 16777216) % 256) end
  local function le16(x) return char(x % 256, floor(x / 256) % 256) end
  local cab = "RIFF" .. le32(36 + #datos) .. "WAVEfmt " .. le32(16) .. le16(1) .. le16(1) .. le32(rate) .. le32(rate * 2) .. le16(2) .. le16(16) .. "data" .. le32(#datos)
  local f = io.open(snd_path .. "/" .. id .. ".wav", "wb")
  if not f then return false end
  f:write(cab .. datos)
  f:close()
  return true
end

local PI, sin, exp = math.pi, math.sin, math.exp
local function ruido() return math.random() * 2 - 1 end

-- Melodía en bucle (notas con 0 = silencio) con percusión opcional
local function melodia(notas, paso, vol, perc)
  return function(t)
    local k = math.floor(t / paso) % #notas + 1
    local tt = t % paso
    local f = notas[k]
    local v = 0
    if f > 0 then v = (sin(2 * PI * f * tt) + 0.35 * sin(4 * PI * f * tt)) * exp(-tt * 5) * vol end
    if perc then v = v + ruido() * exp(-(t % perc) * 40) * 0.12 end
    return v
  end
end

-- Arpegio de una sola vez (fanfarrias)
local function arpegio(notas, paso)
  return function(t)
    local k = math.min(#notas, math.floor(t / paso) + 1)
    local tt = t - (k - 1) * paso
    local f = notas[k]
    return (sin(2 * PI * f * tt) + 0.4 * sin(4 * PI * f * tt)) * exp(-tt * 6) * 0.45
  end
end

    for id, _ in pairs(SINTETICOS) do
      pcall(function() File(snd_path .. "/" .. id .. ".wav").delete() end)
    end
    config.sonidos_v = 2

    if not rutaSonido(id) then

local function fusionar(pt, def)
  for k, v in pairs(def) do
    if pt[k] == nil then
      pt[k] = v
    elseif type(v) == "table" and type(pt[k]) == "table" and k ~= "inventario_armas" then
      fusionar(pt[k], v)
    end
  end
end

local function enteros(t)
  for k, v in pairs(t) do
    if type(v) == "number" then t[k] = math.floor(v)
    elseif type(v) == "table" then enteros(v) end
  end
end

local function recalcularVida(pt)
  pt.vida_max = 100 + (pt.rango - 1) * 8 + pt.hab.resistencia * 15
end

  end

  pt.vida = math.max(0, math.min(pt.vida, pt.vida_max))

  pt.nivel_actual = math.max(1, math.min(100, pt.nivel_actual))
  return pt
end

local function datosArma(pt, inv)
  local b = ARMA_POR_NOMBRE[inv.nombre] or ARMAS[1]
  local lvl = pt.mejoras[b.nombre] or 0
  return {
    nombre = b.nombre, dano = ent(b.dano * (1 + 0.15 * lvl)), disparos = b.disparos or 1,
    crit = b.crit or 0, aturdir = b.aturdir or 0, perfora = b.perfora, jefe = b.jefe or 0,
    nivel = lvl, rasgo = b.rasgo, snd = b.snd,
  }
end

local function armaActual(pt)
  local inv = pt.inventario_armas[pt.arma_equipada_idx] or pt.inventario_armas[1]

end

local function descArma(a)
  local s
  if a.disparos > 1 then s = a.disparos .. " disparos de " .. a.dano else s = "daño " .. a.dano end
  if a.nivel > 0 then s = s .. ", mejora " .. a.nivel end
  return s
end

local function tieneArma(pt, nombre)
  for _, inv in ipairs(pt.inventario_armas) do
    if inv.nombre == nombre then return true end
  end
  return false
end

  end
  return msg
end

local function esquivaJugador(pt)
  return math.min(60, 5 + pt.hab.reflejos * 3 + (pt.caballo and 7 or 0))
end

local function registrarBaja(pt, en)
  pt.stats.kills = pt.stats.kills + 1
  if en.jefe then pt.stats.jefes = pt.stats.jefes + 1 end
end

local function ataqueEnemigo(pt, c, en, factor)
  if en.aturdido then
    en.aturdido = false
    return 0, en.nombre .. " está aturdido y pierde su turno."
  end
  local dano = en.ataque * factor * (0.85 + math.random() * 0.3)
  local esp = nil
  local r = math.random(100)
  if en.tipo == "bruto" and r <= 20 then dano = dano * 1.6 esp = "¡golpe aplastante!"
  elseif en.tipo == "dinamitero" and r <= 25 then dano = dano * 1.8 esp = "¡lanza dinamita!"
  elseif en.tipo == "francotirador" and r <= 25 then dano = dano * 1.7 esp = "¡disparo certero!"
  elseif en.tipo == "jefe" and r <= 25 then dano = dano * 1.6 esp = "¡ataque devastador!"
  elseif en.tipo == "cazasombras" and r <= 20 then dano = dano * 1.6 esp = "¡ataque a traición!"
  elseif en.tipo == "titan" and r <= 15 then dano = dano * 1.7 esp = "¡pisotón demoledor!"
  elseif en.tipo == "espectro" and r <= 25 then dano = dano * 1.5 esp = "¡toque helado!" end
  if en.furia then dano = dano * 1.4 end
  if math.random(100) <= esquivaJugador(pt) then
    reproducirEfecto("snd_esquiva", true)
    return 0, "Esquivas el ataque de " .. en.nombre .. "."
  end
  if c.cubierto then dano = dano * 0.4 end
  dano = math.max(1, ent(dano * (1 - 0.02 * pt.armadura)) - pt.armadura * 3)
  return dano, en.nombre .. " te ataca" .. (esp and (": " .. esp) or ".")
end

local function disparoJugador(pt, c, en, arma, apuntado)
  local esq = en.esq or 0
  if apuntado or arma.perfora then esq = 0 end
  if math.random(100) <= esq then return 0, false, true end
  local d = arma.dano * (0.9 + math.random() * 0.2)
  if c.whisky > 0 then d = d * 1.3 end
  if en.jefe and arma.jefe > 0 then d = d * (1 + arma.jefe) end
  local prob = math.min(85, 8 + pt.hab.punteria * 4 + arma.crit)
  local crit = apuntado or (math.random(100) <= prob)
  if crit then d = d * 2 end
  return math.max(1, ent(d)), crit, false
end

function mostrarCombate(msg)
  local pt, c = p(), combate
  if not c then mostrarMenu() return end
  local lineas = {}
  for i, e in ipairs(c.enemigos) do
    table.insert(lineas, (i == 1 and "▶ " or "• ") .. e.nombre .. ": vida " .. e.vida .. "/" .. e.vida_max .. (e.furia and " [EN FURIA]" or ""))
  end
  local estados = ""
  if c.whisky > 0 then estados = estados .. "\n🥃 Whisky activo: " .. c.whisky .. " turno(s)." end
  if c.apuntando then estados = estados .. "\n🎯 Apuntando: el próximo disparo será crítico." end
  mostrarLayout(c.titulo, (msg or "") .. "\n\nEnemigos:\n" .. table.concat(lineas, "\n") .. estados .. "\n\n" .. hud(), botonesCombate())
  narrar((msg or "") .. " Tu vida: " .. pt.vida .. ". Enemigo al frente: " .. c.enemigos[1].nombre .. ", vida " .. c.enemigos[1].vida .. ".")
end

function finTurno(partes)
  local pt, c = p(), combate
  if not c then mostrarMenu() return end
  if #c.enemigos == 0 then victoriaCombate(partes) return end

  local frente = c.enemigos[1]
  if frente.jefe and not frente.furia and frente.vida <= frente.vida_max * 0.5 then
    frente.furia = true
    table.insert(partes, "¡" .. frente.nombre .. " entra en FURIA! Ahora golpea más fuerte.")
  end

  local recibido, extra = 0, 0
  for i, en in ipairs(c.enemigos) do
    local d, txt = ataqueEnemigo(pt, c, en, (i == 1) and 1 or 0.3)
    if i == 1 then
      if txt then table.insert(partes, txt) end
      recibido = recibido + d
    else
      extra = extra + d
    end
  end
  if extra > 0 then
    table.insert(partes, "Los demás enemigos disparan desde atrás.")
    recibido = recibido + extra
  end
  pt.vida = pt.vida - recibido
  c.cubierto = false
  if c.whisky > 0 then
    c.whisky = c.whisky - 1
    if c.whisky == 0 then table.insert(partes, "Se te pasa el efecto del whisky.") end
  end
  if recibido > 0 then
    reproducirEfecto("snd_golpe", true)
    table.insert(partes, "Recibes " .. recibido .. " de daño.")
  end
  if pt.vida <= 0 then
    guardarPartida()
    muerte(table.concat(partes, "\n"))
    return
  end
  guardarPartida()
  c.turno = c.turno + 1
  mostrarCombate(table.concat(partes, "\n"))
end

local function accionDisparar()
  local pt, c = p(), combate
  if not c then mostrarMenu() return end
  local arma = armaActual(pt)
  reproducirEfecto(arma.snd or "snd_revolver")

  local total, crits, fallos, impactos = 0, 0, 0, 0
  local caidos = {}
  local aturdido = false
  for s = 1, arma.disparos do
    local en = c.enemigos[1]
    if not en then break end
    local d, crit, fallo = disparoJugador(pt, c, en, arma, (s == 1) and c.apuntando)
    if fallo then
      fallos = fallos + 1
    else
      impactos = impactos + 1
      en.vida = en.vida - d
      total = total + d
      if crit then crits = crits + 1 pt.stats.criticos = pt.stats.criticos + 1 end
      if en.vida <= 0 then
        table.insert(caidos, en.nombre)
        registrarBaja(pt, en)
        table.remove(c.enemigos, 1)
      elseif arma.aturdir > 0 and math.random(100) <= arma.aturdir then
        en.aturdido = true
        aturdido = true
      end
    end
  end
  c.apuntando = false
  pt.stats.dano_total = pt.stats.dano_total + total
  if crits > 0 then reproducirEfecto("snd_critico", true) end

  local msg = "Disparas con " .. arma.nombre .. ": "
  if arma.disparos == 1 then
    if fallos > 0 then msg = msg .. "¡fallas, el enemigo esquiva!"
    else msg = msg .. total .. " de daño" .. (crits > 0 and ". ¡GOLPE CRÍTICO!" or ".") end
  else
    msg = msg .. impactos .. " de " .. arma.disparos .. " disparos aciertan, " .. total .. " de daño" .. (crits > 0 and (", " .. crits .. " crítico(s)") or "") .. "."
  end
  local partes = { msg }
  if #caidos > 0 then table.insert(partes, "Abates a: " .. table.concat(caidos, ", ") .. ".") end
  if aturdido then table.insert(partes, "¡El impacto deja aturdido al enemigo!") end
  finTurno(partes)
end

  if pt.dinamitas <= 0 then hablar("No tienes dinamita.") return end
  pt.dinamitas = pt.dinamitas - 1
  pt.stats.dinamitas_usadas = pt.stats.dinamitas_usadas + 1
  reproducirEfecto("snd_dinamita")
  local basedano = 55 + pt.rango * 5 + pt.hab.punteria * 2
  local vivos, total, caidos = {}, 0, 0
  for _, en in ipairs(c.enemigos) do
    local d = ent(basedano * (0.9 + math.random() * 0.2) * (en.jefe and 0.7 or 1))
    en.vida = en.vida - d
    total = total + d
    if en.vida > 0 then
      table.insert(vivos, en)
    else
      caidos = caidos + 1
      registrarBaja(pt, en)
    end
  end
  c.enemigos = vivos
  pt.stats.dano_total = pt.stats.dano_total + total
  finTurno({ "¡Lanzas dinamita! La explosión causa " .. total .. " de daño en total y derriba a " .. caidos .. " enemigo(s)." })
end

local function accionWhisky()
  local pt, c = p(), combate
  if not c then mostrarMenu() return end
  if pt.whisky <= 0 then hablar("No tienes whisky.") return end
  pt.whisky = pt.whisky - 1
  c.whisky = 4
  finTurno({ "Te echas un trago de whisky: +30% de daño durante 4 turnos." })
end

local function accionBotiquin()
  local pt, c = p(), combate
  if not c then mostrarMenu() return end
  if pt.botiquines <= 0 then hablar("No tienes botiquines.") return end
  pt.botiquines = pt.botiquines - 1
  local cura = math.min(curaBot(pt), pt.vida_max - pt.vida)
  pt.vida = pt.vida + cura
  reproducirEfecto("snd_curar", true)
  finTurno({ "Usas un botiquín y recuperas " .. cura .. " de vida. El enemigo aprovecha el momento." })
end

local function accionHuir()
  local pt, c = p(), combate
  if not c then mostrarMenu() return end
  local prob = 45 + (pt.caballo and 30 or 0) + pt.hab.reflejos * 2
  if c.jefe then prob = ent(prob / 2) end
  prob = math.min(90, prob)
  if math.random(100) <= prob then
    local fn = c.alHuir
    combate = nil
    hablar("Escapaste con vida.")
    fn()
  else
    finTurno({ "Intentas huir, pero te cierran el paso." })
  end
end

  end)

function botonesCombate()
  local pt, c = p(), combate
  local arma = armaActual(pt)
  local bots = {
    { "Disparar con " .. arma.nombre .. " (" .. descArma(arma) .. ")", accionDisparar },
    { "Apuntar: próximo disparo crítico seguro", accionApuntar },
    { "Cubrirse: reduce 60% el daño este turno", accionCubrirse },
  }
  if pt.dinamitas > 0 then table.insert(bots, { "Lanzar dinamita (" .. pt.dinamitas .. "): daño a todos", accionDinamita }) end
  if pt.whisky > 0 and c.whisky == 0 then table.insert(bots, { "Beber whisky (" .. pt.whisky .. "): +30% daño", accionWhisky }) end
  if pt.botiquines > 0 and pt.vida < pt.vida_max then

  end
  if #pt.inventario_armas > 1 then table.insert(bots, { "Cambiar de arma (gratis)", accionCambiarArma }) end

local function campanaVencida(txt)
  local pt, v = p(), viaje
  if not v then mostrarMenu() return end
  local n = v.nivel
  local final = v.etapa >= v.etapas
  local d, xp
  if final then
    d = 8 + n * 2 + math.random(0, 5)
    xp = 30 + n * 12
    if v.jefe_final then d = d + 15 + n xp = xp + 40 end
  else
    d = math.random(1, 3) + math.floor(n / 3)
    xp = 8 + n * 2
  end
  local real = ganarDiamantes(pt, d)
  txt = txt .. "\n💎 +" .. real .. " diamantes, ✨ +" .. xp .. " XP." .. dar_xp(pt, xp)

  if not final then
    guardarPartida()
    pausaViaje(txt)
    return
  end

  if n >= pt.nivel_actual and pt.nivel_actual < MAX_NIVELES then
    pt.nivel_actual = pt.nivel_actual + 1
    txt = txt .. "\n🔓 ¡Desbloqueas el nivel " .. pt.nivel_actual .. "!"
  end
  if n == MAX_NIVELES then pt.campana_completa = true txt = txt .. "\n👑 ¡Has completado toda la campaña!" end
  txt = txt .. revisarLogros(pt)
  viaje = nil
  guardarPartida()
  cambiarMusica("lobby_music")
  local bots = {}
  if n < MAX_NIVELES and n + 1 <= pt.nivel_actual then
    table.insert(bots, { "Jugar el nivel " .. (n + 1), function() iniciarNivel(n + 1) end })
  end
  table.insert(bots, { "Volver al menú", mostrarMenu })
  mostrarLayout("¡Nivel " .. n .. " superado!", txt .. "\n\n" .. hud(), bots)
  hablar("¡Nivel " .. n .. " superado!")
end

function pausaViaje(txt)
  local pt, v = p(), viaje
  if not v then mostrarMenu() return end
  local bots = { { "Continuar explorando (etapa " .. math.min(v.etapas, v.etapa + 1) .. " de " .. v.etapas .. ")", avanzarEtapa } }
  if pt.botiquines > 0 and pt.vida < pt.vida_max then
    table.insert(bots, { "Usar botiquín (" .. pt.botiquines .. ")", function()
        pt.botiquines = pt.botiquines - 1
        local cura = math.min(curaBot(pt), pt.vida_max - pt.vida)
        pt.vida = pt.vida + cura
        reproducirEfecto("snd_curar", true)
        guardarPartida()
        pausaViaje("Te curas " .. cura .. " de vida.")
      end })
  end
  table.insert(bots, { "Retirarme al pueblo", function() viaje = nil mostrarMenu() end })
  mostrarLayout("Nivel " .. v.nivel .. " - Etapa " .. v.etapa .. " de " .. v.etapas, (txt or "") .. "\n\n" .. hud(), bots)
  hablar(txt or "Continúas tu camino.")
end

function comenzarCombateNivel(final, intro_ev)
  local v = viaje
  if not v then mostrarMenu() return end
  local n = v.nivel
  local hp, atk = escala(n)
  local enemigos, jefe, intro = {}, false, nil
  if final then
    local nombre_jefe = jefeDe(n)
    if nombre_jefe then
      jefe = true
      table.insert(enemigos, crearJefe(nombre_jefe, hp * 3.0, atk * 1.25))
      if n >= 10 then table.insert(enemigos, crearEnemigo(tipoAleatorio(n), hp, atk)) end
      intro = "¡Aparece el jefe " .. nombre_jefe .. "! Prepárate."
    else
      local cnt = (n >= 60) and 4 or ((n >= 10) and 3 or ((n >= 5) and 2 or 1))
      for _ = 1, cnt do table.insert(enemigos, crearEnemigo(tipoAleatorio(n), hp * 1.15, atk * 1.05)) end
      intro = "¡Enfrentamiento final del nivel! Hay " .. cnt .. " enemigo(s) de élite."
    end
  else
    local cnt = 1 + ((n >= 4) and math.random(0, 1) or 0) + ((n >= 10) and math.random(0, 1) or 0)
    for _ = 1, cnt do table.insert(enemigos, crearEnemigo(tipoAleatorio(n), hp, atk)) end
    intro = intro_ev or ("¡Te salen al paso " .. cnt .. " enemigo(s)!")
  end
  v.jefe_final = jefe
  iniciarCombate({
    tipo = "campana", titulo = "Nivel " .. n .. " - Etapa " .. v.etapa .. "/" .. v.etapas,
    enemigos = enemigos, jefe = jefe, intro = intro,
    alVencer = campanaVencida,
    alHuir = function() viaje = nil cambiarMusica("lobby_music") mostrarMenu() end,
  })
end

-- ============================================================
local function hordaVencida(txt)
  local pt, h = p(), horda
  if not h then mostrarMenu() return end
  local n = h.oleada
  local d = 4 + n * 2
  local xp = 12 + n * 4
  if n % 5 == 0 then d = d + 10 + n xp = xp + 30 end
  local etiqueta, curapct = "Oleada", 0.15
  if h.modo == "torre" then
    d = 10 + n * 3
    xp = 20 + n * 6
    etiqueta, curapct = "Piso", 0.25
    if n > pt.stats.torre_max then pt.stats.torre_max = n end
  elseif n > pt.stats.mejor_oleada then
    pt.stats.mejor_oleada = n
  end
  h.botin_d = h.botin_d + d
  h.botin_xp = h.botin_xp + xp
  h.superadas = n
  local cura = math.min(pt.vida_max - pt.vida, ent(pt.vida_max * curapct))
  pt.vida = pt.vida + cura
  guardarPartida()

  local function pantalla(t)
    local bots = { { "Siguiente " .. string.lower(etiqueta) .. " (" .. (n + 1) .. ")", siguienteOleada } }
    if pt.botiquines > 0 and pt.vida < pt.vida_max then
      table.insert(bots, { "Usar botiquín (" .. pt.botiquines .. ")", function()
          pt.botiquines = pt.botiquines - 1
          local c2 = math.min(curaBot(pt), pt.vida_max - pt.vida)
          pt.vida = pt.vida + c2
          reproducirEfecto("snd_curar", true)
          guardarPartida()
          pantalla("Te curas " .. c2 .. " de vida.")
        end })
    end
    table.insert(bots, { "Retirarme con el botín (" .. h.botin_d .. " diamantes, " .. h.botin_xp .. " XP)", retirarHorda })
    mostrarLayout(etiqueta .. " " .. n .. " superado", t .. "\n\nBotín acumulado: " .. h.botin_d .. " diamantes y " .. h.botin_xp .. " XP.\nSi mueres, solo conservarás el 40%.\n\n" .. hud(), bots)
  end

  pantalla(txt .. "\nRecuperas " .. cura .. " de vida entre rondas.")
  hablar(etiqueta .. " " .. n .. " superado. Botín acumulado: " .. h.botin_d .. " diamantes.")
end

function siguienteOleada()
  local pt, h = p(), horda
  if not h then mostrarMenu() return end
  h.oleada = h.oleada + 1
  local n = h.oleada
  if h.modo == "torre" then MD.pisoTorre(h, n) return end
  local hp = 40 + n * 14
  local atk = 9 + n * 2.2
  local enemigos, jefe, intro = {}, false, nil
  if n % 5 == 0 then
    jefe = true
    local nombre = JEFES_HORDA[(math.floor(n / 5) - 1) % #JEFES_HORDA + 1]
    table.insert(enemigos, crearJefe(nombre, hp * 3.0, atk * 1.25))
    for _ = 1, math.min(2, math.floor(n / 10)) do table.insert(enemigos, crearEnemigo(tipoAleatorio(n + 1), hp, atk)) end
    intro = "¡Oleada " .. n .. "! Llega el jefe " .. nombre .. "."
  else
    local cnt = math.min(4, 1 + math.floor(n / 3))
    for _ = 1, cnt do table.insert(enemigos, crearEnemigo(tipoAleatorio(n + 1), hp, atk)) end
    intro = "¡Oleada " .. n .. "! Se acercan " .. cnt .. " enemigo(s)."
  end
  iniciarCombate({
    tipo = "horda", titulo = "Supervivencia - Oleada " .. n,
    enemigos = enemigos, jefe = jefe, intro = intro,
    alVencer = hordaVencida,
    alHuir = function() retirarHorda() end,
  })
end

function retirarHorda()
  local pt, h = p(), horda
  horda = nil
  combate = nil
  if not h then mostrarMenu() return end
  local d = ganarDiamantes(pt, h.botin_d)
  local txt = "Te retiras con vida tras superar " .. (h.superadas or 0) .. (h.modo == "torre" and " piso(s)" or " oleada(s)") .. ".\n💎 +" .. d .. " diamantes, ✨ +" .. h.botin_xp .. " XP." .. dar_xp(pt, h.botin_xp)
  txt = txt .. revisarLogros(pt)
  guardarPartida()
  cambiarMusica("lobby_music")
  mostrarLayout("Fin de la supervivencia", txt .. "\n\n" .. hud(), { { "Volver al menú", mostrarMenu } })
  hablar("Te retiras con el botín.")
end

local function aplicarInteres(pt)
  if pt.banco <= 0 or pt.banco_t == 0 then pt.banco_t = os.time() return 0 end
  local dias = math.floor((os.time() - pt.banco_t) / 86400)
  if dias < 1 then return 0 end
  dias = math.min(dias, 10)
  local gan = math.min(40 * dias, math.floor(pt.banco * 0.03 * dias))
  pt.banco_t = os.time()
  pt.banco = pt.banco + gan
  return gan
end

function mostrarBanco(msg)
  local pt = p()
  local gan = aplicarInteres(pt)
  if gan > 0 then
    msg = (msg and (msg .. "\n") or "") .. "💰 Tu dinero ha generado " .. gan .. " diamantes de interés (3% diario)."
    guardarPartida()
  end
  local function depositar(c)
    return function()
      local cant = (c == 0) and pt.diamantes or c
      if cant <= 0 or pt.diamantes < cant then hablar("No tienes suficientes diamantes.") return end
      if pt.banco == 0 then pt.banco_t = os.time() end
      pt.diamantes = pt.diamantes - cant
      pt.banco = pt.banco + cant
      local lg = revisarLogros(pt)
      guardarPartida()
      mostrarBanco("Depositaste " .. cant .. " diamantes." .. lg)
    end
  end
  mostrarLayout("Banco del Pueblo", (msg and (msg .. "\n\n") or "") .. "En tu cuenta: " .. pt.banco .. " diamantes.\nEn tu bolsa: " .. pt.diamantes .. " diamantes.\nEl banco paga 3% de interés por cada día real (máx. 40 por día).", {
    { "Depositar 25", depositar(25) },
    { "Depositar 100", depositar(100) },
    { "Depositar todo", depositar(0) },
    { "Retirar todo", function()
        if pt.banco <= 0 then hablar("No tienes nada en el banco.") return end
        local cant = pt.banco
        pt.banco = 0
        pt.diamantes = pt.diamantes + cant
        guardarPartida()
        mostrarBanco("Retiraste " .. cant .. " diamantes.")
      end },
    { "Volver", mostrarMenu },
  })
  hablar(msg or ("Banco. Tienes " .. pt.banco .. " diamantes ahorrados."))
end

      local gastados = 0
      for _, h in ipairs(HABS) do gastados = gastados + pt.hab[h.id] end

      pt.vida = math.min(pt.vida, pt.vida_max)
      guardarPartida()

function recompensaDiaria()
  local pt = p()
  local hoy = os.date("%Y%m%d")
  if pt.ultima_recompensa == hoy then
    mostrarLayout("Recompensa diaria", "Ya reclamaste la recompensa de hoy. Vuelve mañana.\nRacha actual: " .. pt.racha .. " día(s).", { { "Volver", mostrarMenu } })
    hablar("Ya reclamaste la recompensa de hoy.")
    return
  end
  local ayer = os.date("%Y%m%d", os.time() - 86400)
  if pt.ultima_recompensa == ayer then pt.racha = pt.racha + 1 else pt.racha = 1 end
  pt.ultima_recompensa = hoy
  local d = 10 + math.min(pt.racha, 7) * 5
  sumarDiamantes(pt, d)
  local txt = "🎁 ¡Recompensa del día " .. pt.racha .. " de racha!\n+" .. d .. " diamantes"
  if pt.racha >= 3 then pt.botiquines = math.min(10, pt.botiquines + 1) txt = txt .. ", +1 botiquín" end
  if pt.racha >= 5 then pt.dinamitas = math.min(10, pt.dinamitas + 1) txt = txt .. ", +1 dinamita" end
  if pt.racha >= 7 then pt.whisky = math.min(10, pt.whisky + 1) txt = txt .. ", +1 whisky" end
  txt = txt .. ", +20 XP." .. dar_xp(pt, 20) .. revisarLogros(pt)
  guardarPartida()
  reproducirEfecto("snd_moneda")
  mostrarLayout("Recompensa diaria", txt .. "\n\nCuantos más días seguidos juegues, mejores premios.", { { "Volver", mostrarMenu } })
  hablar("Recompensa diaria reclamada. Racha de " .. pt.racha .. " días. Ganaste " .. d .. " diamantes.")
end

function cambiarPassword()
  local ids = {}
  local cont = {
    LinearLayout, orientation = "vertical", layout_width = "fill", padding = "14dp",
    { TextView, text = "Contraseña actual:", layout_width = "fill" },
    { EditText, id = "c_actual", hint = "Contraseña actual", inputType = 0x81, layout_width = "fill", layout_marginBottom = "10dp" },
    { TextView, text = "Contraseña nueva:", layout_width = "fill" },
    { EditText, id = "c_nueva", hint = "Mínimo 3 caracteres", inputType = 0x81, layout_width = "fill", layout_marginBottom = "15dp" },
    { Button, text = "Cambiar contraseña", layout_width = "fill", layout_marginBottom = "6dp", onClick = function()
        local actual = trim(tostring(ids.c_actual.getText()))
        local nueva = trim(tostring(ids.c_nueva.getText()))
        local db = cargarUsuarios()
        local cuenta = db[session.usuario]
        if not cuenta or (cuenta.hash ~= hashPass(actual, cuenta.sal) and cuenta.pass ~= actual) then
          hablar("La contraseña actual es incorrecta.") return
        end
        if #nueva < 3 then hablar("La nueva contraseña debe tener al menos 3 caracteres.") return end
        cuenta.sal = nuevaSal()
        cuenta.hash = hashPass(nueva, cuenta.sal)
        cuenta.pass = nil
        guardarUsuarios(db)
        hablar("Contraseña cambiada con éxito.")
        mostrarAjustes()
      end },
    { Button, text = "Volver", layout_width = "fill", onClick = function() mostrarAjustes() end },
  }
  cerrar()
  local d = LuaDialog(ctx)
  d.setTitle("Cambiar contraseña")
  d.setView(loadlayout({ ScrollView, layout_width = "fill", cont }, ids))
  d.show()
  dlg_activo = d
end

    return vols[1]

    "Música: " .. (config.musica and "activada" or "desactivada") ..

      end },

        mostrarAjustes("Volumen al " .. ent(config.volumen * 100) .. "%.")
      end },
    { "Narración detallada: " .. (config.narracion ~= false and "desactivar" or "activar"), function()
        config.narracion = not (config.narracion ~= false)
        guardarConfig()
        mostrarAjustes("Narración " .. (config.narracion and "activada." or "desactivada."))
      end },
    { "Cambiar contraseña", cambiarPassword },
    { "Novedades de la versión 3.0", function() mostrarNovedades(mostrarAjustes) end },

-- ---------- Mina de oro ----------
function MD.mina(msg)
  local pt = p()
  local m = pt.mina
  local function pendiente()
    if m.nivel <= 0 or m.t == 0 then return 0 end
    return math.floor(math.min(12, (os.time() - m.t) / 3600) * m.nivel * 3)
  end
  local bots = {}
  if m.nivel <= 0 then
    table.insert(bots, { "Comprar la mina - 200 diamantes", function()
        if pt.diamantes < 200 then hablar("Diamantes insuficientes.") return end
        pt.diamantes = pt.diamantes - 200
        m.nivel = 1
        m.t = os.time()
        local lg = revisarLogros(pt)
        guardarPartida()
        MD.mina("¡Compraste la mina! Producirá diamantes aunque no estés jugando." .. lg)
      end })
  else
    table.insert(bots, { "Recoger " .. pendiente() .. " diamantes", function()
        local ahora = pendiente()
        if ahora <= 0 then hablar("Todavía no hay nada que recoger.") return end
        sumarDiamantes(pt, ahora)
        m.t = os.time()
        reproducirEfecto("snd_moneda")
        guardarPartida()
        MD.mina("Recogiste " .. ahora .. " diamantes.")
      end })
    if m.nivel < 25 then
      local costo = ent(50 * (m.nivel + 1) ^ 1.7)
      table.insert(bots, { "Mejorar la mina al nivel " .. (m.nivel + 1) .. " - " .. costo .. " diamantes", function()
          if pt.diamantes < costo then hablar("Diamantes insuficientes.") return end
          local ahora = pendiente()
          if ahora > 0 then sumarDiamantes(pt, ahora) end
          pt.diamantes = pt.diamantes - costo
          m.nivel = m.nivel + 1
          m.t = os.time()
          local lg = revisarLogros(pt)
          guardarPartida()
          reproducirEfecto("snd_nivel", true)
          MD.mina("Mina mejorada al nivel " .. m.nivel .. "." .. lg)
        end })
    end
  end
  table.insert(bots, { "Volver al pueblo", function() MD.pueblo() end })
  mostrarLayout("Mina de Oro", (msg and (msg .. "\n\n") or "") .. "Nivel de la mina: " .. m.nivel .. " de 25.\nProduce " .. (m.nivel * 3) .. " diamantes por hora y guarda hasta 12 horas.\nDiamantes: " .. pt.diamantes, bots)
  hablar(msg or ("Mina de oro, nivel " .. m.nivel .. "."))
end

if ON.srv < 1 or ON.srv > #ON.SERVERS then ON.srv = 1 end
ON.token = config.on_token or ""
ON.nombre = config.on_name or ""
ON.ultimo = 0

function ON.req(metodo, ruta, datos, cb)
  Thread(Runnable{ run = function()
    local code, texto, ok = 0, "", false
    for intento = 0, #ON.SERVERS - 1 do
      local idx = ((ON.srv - 1 + intento) % #ON.SERVERS) + 1
      code, texto = 0, ""
      ok = pcall(function()
        local c = URL(ON.SERVERS[idx] .. ruta).openConnection()
        c.setRequestMethod(metodo)
        c.setConnectTimeout(4000)
        c.setReadTimeout(12000)
        c.setRequestProperty("Content-Type", "application/json")
        if ON.token ~= "" then c.setRequestProperty("Authorization", "Bearer " .. ON.token) end
        if datos then
          c.setDoOutput(true)
          local w = OutputStreamWriter(c.getOutputStream(), "UTF-8")
          w.write(cjson.encode(datos))
          w.flush()
          w.close()
        end
        code = c.getResponseCode()
        local stream = (code >= 400) and c.getErrorStream() or c.getInputStream()
        local br = BufferedReader(InputStreamReader(stream, "UTF-8"))
        local partes = {}
        local linea = br.readLine()
        while linea do partes[#partes + 1] = linea; linea = br.readLine() end
        br.close()
        texto = table.concat(partes, "\n")
      end)
      if ok and code < 500 then ON.srv = idx; config.srv = idx; break end
    end
    handler.post(Runnable{ run = function()
      if not ok then cb(0, nil, "No se pudo conectar con ningún servidor. Revisa tu internet.") return end
      local pok, obj = pcall(cjson.decode, texto)
      if not pok or type(obj) ~= "table" then cb(code, nil, "Respuesta no válida del servidor (código " .. code .. ").") return end
      cb(code, obj, nil)
    end })
  end }).start()
end

function ON.msgError(code, o, err)
  if err then return err end
  if code == 401 then
    ON.token = ""
    config.on_token = nil
    guardarConfig()
    return "Tu sesión en línea terminó (quizá entraste desde otro teléfono). Entra de nuevo."
  end
  return (o and o.error) or "Error desconocido."
end

-- Cobra premios pendientes de duelos (cb(ok, texto, cantidad))
function ON.reclamar(cb)
  ON.req("POST", "/api/claim", {}, function(code, o, err)
    if err or not o.ok then cb(false, ON.msgError(code, o, err), 0) return end
    local pt = p()
    local txt, n = "", 0
    for _, m in ipairs(o.mail or {}) do
      n = n + 1
      local d = math.floor(tonumber(m.diamantes) or 0)
      local xp = math.floor(tonumber(m.xp) or 0)
      local t = tostring(m.texto or "")
      if d > 0 then sumarDiamantes(pt, d) end
      txt = txt .. "\n• " .. t
      if d > 0 or xp > 0 then txt = txt .. " (+" .. d .. " diamantes, +" .. xp .. " XP)" end
      if xp > 0 then txt = txt .. dar_xp(pt, xp) end
      if string.find(t, "ganaste", 1, true) then pt.stats.duelos_ganados = pt.stats.duelos_ganados + 1 end
    end
    if n > 0 then txt = txt .. revisarLogros(pt) guardarPartida() end
    cb(true, n > 0 and ("Novedades de tus duelos:" .. txt) or "No tienes novedades.", n)
  end)
end

function ON.autosync()
  ON.ultimo = os.time()
  ON.subir(function(ok)
    if ok then
      ON.reclamar(function(ok2, txt, n) if n and n > 0 then hablar(txt) end end)
    end
  end)
end

function ON.menu(msg)
  local conectado = ON.token ~= ""
  local bots = {}
  if not conectado then
    table.insert(bots, { "Entrar a mi cuenta en línea", function() ON.entrar(false) end })
    table.insert(bots, { "Crear cuenta en línea", function() ON.entrar(true) end })
  else
    table.insert(bots, { "⚔ Duelos contra otros jugadores", function() ON.duelos() end })
    table.insert(bots, { "🏅 Ranking mundial", function() ON.ranking("score") end })
    table.insert(bots, { "💬 Chat del pueblo", function() ON.chat() end })
    table.insert(bots, { "🔎 Buscar jugadores", function() ON.buscarPedir() end })
    table.insert(bots, { "☁ Subir mi partida a la nube ahora", function()
        ON.subir(function(ok, m) ON.menu(m) end)
      end })
    table.insert(bots, { "⬇ Descargar mi partida de la nube", function() ON.bajarConfirmar() end })
    table.insert(bots, { "Cerrar sesión en línea", function()
        ON.token = ""
        config.on_token = nil
        config.on_name = nil
        guardarConfig()
        ON.nombre = ""
        ON.menu("Sesión en línea cerrada. Tu partida local sigue en este teléfono.")
      end })
  end
  table.insert(bots, { "Probar los servidores", function() ON.probar() end })
  table.insert(bots, { "Volver", mostrarMenu })
  local estado = conectado and ("Conectado como " .. ON.nombre .. ". Tu partida se guarda sola en la nube cada pocos minutos.")
    or "Crea una cuenta en línea para guardar tu partida en la nube, aparecer en el ranking mundial, retar a otros jugadores y hablar en el chat."
  mostrarLayout("Juego en línea", (msg and (msg .. "\n\n") or "") .. estado, bots)
  hablar(msg or "Juego en línea.")
end

function ON.entrar(registro)
  ON.pedir(registro and "Crear cuenta en línea" or "Entrar a mi cuenta en línea",
    { "Nombre de usuario (3 a 20 letras)", "Contraseña (mínimo 4)" }, true, function(v)
      if v[1] == "" or v[2] == "" then hablar("Escribe usuario y contraseña.") ON.entrar(registro) return end
      ON.token = ""
      ON.req("POST", registro and "/api/register" or "/api/login", { name = v[1], password = v[2] }, function(code, o, err)
        if err or not o.ok then ON.menu(ON.msgError(code, o, err)) return end
        ON.token = o.token
        ON.nombre = o.name
        config.on_token = o.token
        config.on_name = o.name
        guardarConfig()
        ON.sincronizarInicial()
      end)
    end, function() ON.menu() end)
end

function ON.sincronizarInicial()
  ON.req("GET", "/api/state", nil, function(code, o, err)
    if err or not o.ok then ON.menu(ON.msgError(code, o, err)) return end
    if type(o.partida) ~= "table" then
      ON.subir(function(ok, m) ON.menu("Cuenta lista. " .. m) end)
      return
    end
    local nube, pt = o.partida, p()
    mostrarLayout("Partida en la nube", "Tu cuenta en línea ya tiene una partida: rango " .. tostring(nube.rango or 1) .. ", nivel " .. tostring(nube.nivel_actual or 1) .. ".\nTu partida en este teléfono: rango " .. pt.rango .. ", nivel " .. pt.nivel_actual .. ".\n¿Cuál quieres usar?", {
      { "Usar la de la nube (reemplaza la de este teléfono)", function()
          session.partida = normalizarPartida(nube)
          guardarPartida()
          ON.menu("Partida de la nube descargada.")
        end },
      { "Usar la de este teléfono (reemplaza la de la nube)", function()
          ON.subir(function(ok, m) ON.menu(m) end)
        end },
    })
  end)
end

function ON.bajarConfirmar()
  mostrarLayout("Descargar partida", "Se reemplazará la partida de este teléfono por la que está guardada en la nube. ¿Continuar?", {
    { "Sí, descargar", function()
        ON.req("GET", "/api/state", nil, function(code, o, err)
          if err or not o.ok then ON.menu(ON.msgError(code, o, err)) return end
          if type(o.partida) ~= "table" then ON.menu("Todavía no hay partida en la nube.") return end
          session.partida = normalizarPartida(o.partida)
          guardarPartida()
          ON.menu("Partida descargada.")
        end)
      end },
    { "No, volver", function() ON.menu() end },
  })
end

function ON.ranking(tipo)
  ON.req("GET", "/api/ranking?kind=" .. tipo, nil, function(code, o, err)
    if err or not o.ok then ON.menu(ON.msgError(code, o, err)) return end
    local bots = {}
    for i, e in ipairs(o.ranking or {}) do
      table.insert(bots, { i .. ". " .. e.name .. ": rango " .. e.rango .. ", nivel " .. e.nivel .. ", oleada " .. e.oleada .. ", " .. e.score .. " puntos", function() ON.perfil(e.name) end })
    end
    if #bots == 0 then table.insert(bots, { "Todavía no hay jugadores en el ranking.", function() end }) end

    table.insert(bots, { "Volver", function() ON.menu() end })
    mostrarLayout("Ranking mundial", "Toca un jugador para ver su perfil y retarlo a un duelo.", bots)
    hablar("Ranking mundial. " .. #(o.ranking or {}) .. " jugadores.")
  end)
end

function ON.buscarPedir()
  ON.pedir("Buscar jugadores", { "Parte del nombre (vacío = todos)" }, false, function(v) ON.buscar(v[1]) end, function() ON.menu() end)
end

function ON.buscar(q)
  ON.req("GET", "/api/find?q=" .. URLEncoder.encode(q or "", "UTF-8"), nil, function(code, o, err)
    if err or not o.ok then ON.menu(ON.msgError(code, o, err)) return end
    local bots = {}
    for _, e in ipairs(o.players or {}) do
      table.insert(bots, { e.name .. " (rango " .. e.rango .. ")", function() ON.perfil(e.name) end })
    end
    if #bots == 0 then table.insert(bots, { "No se encontró a nadie con ese nombre.", function() end }) end
    table.insert(bots, { "Buscar otra vez", function() ON.buscarPedir() end })
    table.insert(bots, { "Volver", function() ON.menu() end })
    mostrarLayout("Jugadores", "Toca un jugador para ver su perfil.", bots)
  end)
end

function ON.perfil(nombre)
  ON.req("GET", "/api/player?name=" .. URLEncoder.encode(nombre, "UTF-8"), nil, function(code, o, err)
    if err or not o.ok then ON.menu(ON.msgError(code, o, err)) return end

      table.insert(bots, { "⚔ Retar a duelo", function() ON.duelo(o.name) end })

  end)
end

function ON.duelo(nombre)
  hablar("Duelo contra " .. nombre .. ". Preparando.")
  ON.subir(function(ok, m)
    if not ok then ON.menu(m) return end
    ON.req("POST", "/api/duel", { target = nombre }, function(code, o, err)
      if err or not o.ok then ON.duelos(ON.msgError(code, o, err)) return end
      local lineas = {}
      for _, l in ipairs(o.log or {}) do lineas[#lineas + 1] = l end
      ON.reclamar(function(ok2, txt)
        reproducirEfecto(o.won and "victoria" or "snd_derrota")
        mostrarLayout(o.won and "¡Ganaste el duelo!" or "Perdiste el duelo", table.concat(lineas, "\n") .. "\n\n" .. txt .. "\n\n" .. hud(), {
          { "Revancha", function() ON.duelo(nombre) end },
          { "Volver", function() ON.duelos() end },
        })
        hablar(o.won and ("¡Ganaste el duelo contra " .. nombre .. "!") or ("Perdiste el duelo contra " .. nombre .. "."))
      end)
    end)
  end)
end

-- ---------- chat del pueblo ----------
function ON.chat(msg, enviar)
  ON.req(enviar and "POST" or "GET", "/api/chat?after=0", enviar and { text = enviar } or nil, function(code, o, err)
    if err or not o.ok then
      local e = ON.msgError(code, o, err)
      if enviar and code ~= 401 and code ~= 0 then ON.chat(e) else ON.menu(e) end
      return
    end
    local lineas = {}
    for _, m in ipairs(o.messages or {}) do
      lineas[#lineas + 1] = os.date("%H:%M", math.floor((tonumber(m.ts) or 0) / 1000)) .. " " .. tostring(m.name) .. ": " .. tostring(m.text)
    end
    mostrarLayout("Chat del pueblo", (msg and (msg .. "\n\n") or "") .. (#lineas > 0 and table.concat(lineas, "\n") or "Aún no hay mensajes. ¡Escribe el primero!"), {
      { "Escribir un mensaje", function()
          ON.pedir("Mensaje para el pueblo", { "Escribe tu mensaje (máximo 200 letras)" }, false, function(v)
            if v[1] == "" then ON.chat() else ON.chat(nil, v[1]) end
          end, function() ON.chat() end)
        end },
      { "Actualizar", function() ON.chat() end },
      { "Volver", function() ON.menu() end },
    })
    hablar(msg or ("Chat del pueblo. " .. #lineas .. " mensajes."))
  end)
end

-- ---------- prueba de servidores ----------
function ON.probar()
  hablar("Probando los servidores. Puede tardar unos segundos.")
  Thread(Runnable{ run = function()
    local res = {}
    for i, base in ipairs(ON.SERVERS) do
      local t0 = System.currentTimeMillis()
      local code = 0
      local ok = pcall(function()
        local c = URL(base .. "/api/info").openConnection()
        c.setConnectTimeout(3000)
        c.setReadTimeout(5000)
        code = c.getResponseCode()
      end)
      local ms = System.currentTimeMillis() - t0
      local nombre = (i == 1) and "Principal" or ("Respaldo " .. (i - 1))
      if ok and code == 200 then res[#res + 1] = nombre .. ": funciona (" .. ms .. " ms)"
      elseif ok and code == 423 then res[#res + 1] = nombre .. ": cerrado por el administrador"
      elseif code ~= 0 then res[#res + 1] = nombre .. ": error " .. code
      else res[#res + 1] = nombre .. ": sin respuesta" end
    end
    handler.post(Runnable{ run = function()
      ON.menu("Servidor en uso: " .. ((ON.srv == 1) and "Principal" or ("Respaldo " .. (ON.srv - 1))) .. "\n" .. table.concat(res, "\n"))
    end })
  end }).start()
end
